    // --- Simulate Counter ---
    document.getElementById('increment-btn').addEventListener('click', () => {
      const display = document.getElementById('simulator-display');
      let count = parseInt(display.innerText.split(': ')[1]) || 0;
      count++;
      display.innerText = `Count: ${count}`;
    });

    // --- Send Packet Animation ---
    document.getElementById('send-packet-btn').addEventListener('click', () => {
      const btn = document.getElementById('send-packet-btn');
      btn.innerText = 'Sending...';
      btn.disabled = true;
      
      // Create a moving dot
      const svg = document.querySelector('#hdf-visualization svg') || document.querySelector('svg');
      const dot = document.createElementNS("http://www.w3.org/2000/svg", "circle");
      dot.setAttribute("cx", "10%");
      dot.setAttribute("cy", "50%");
      dot.setAttribute("r", "6");
      dot.setAttribute("fill", "#f59e0b");
      svg.appendChild(dot);

      let pos = 10;
      const interval = setInterval(() => {
        pos += 2;
        dot.setAttribute("cx", `${pos}%`);
        if (pos > 90) {
          clearInterval(interval);
          svg.removeChild(dot);
          btn.innerText = 'Send Test Packet';
          btn.disabled = false;
          // Briefly highlight the last node
          const lastNode = svg.querySelector('circle[cx="90%"]');
          lastNode.setAttribute("fill", "#fbbf24");
          setTimeout(() => {
            lastNode.setAttribute("fill", "#f59e0b");
          }, 500);
        }
      }, 50);
    });

    // --- Architecture Interactions ---
    const layers = document.querySelectorAll('#architecture-diagram > div');
    layers.forEach(layer => {
      if (!layer.classList.contains('absolute')) return; // Skip particles
      layer.addEventListener('mouseenter', () => {
        layer.classList.add('glow');
      });
      layer.addEventListener('mouseleave', () => {
        layer.classList.remove('glow');
      });
      layer.addEventListener('click', () => {
        alert(`You clicked on: ${layer.innerText.trim()}`);
      });
    });
  </script>
